# Скопируйте и выполните эти команды на сервере по очереди

tee /etc/nginx/sites-available/zumba-site > /dev/null << 'ENDOFCONFIG'
server {
    listen 80;
    server_name zumba-spb.ru;
    return 301 https://zumba-spb.ru$request_uri;
}

server {
    listen 443 ssl http2;
    server_name zumba-spb.ru;
    
    ssl_certificate /etc/letsencrypt/live/zumba-spb.ru/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/zumba-spb.ru/privkey.pem;
    
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_prefer_server_ciphers on;
    ssl_ciphers HIGH:!aNULL:!MD5;
    
    root /var/www/zumba-site;
    index index.html;
    
    location / {
        try_files $uri $uri/ =404;
    }
}
ENDOFCONFIG

ln -sf /etc/nginx/sites-available/zumba-site /etc/nginx/sites-enabled/

rm -f /etc/nginx/sites-enabled/default

nginx -t

systemctl restart nginx

ss -tlnp | grep -E ':80|:443'
